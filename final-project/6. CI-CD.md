# CI/CD

**Instructions**

[ *CI/CD* ]

- Using Jenkins, create a pipeline running:
   - Repository pull
   - Image build
   - Testing Your Code
   - Push Image into your own docker registry private
   - SSH into your server
   - Pull image from docker registry private
   - Redeploy your deployment apps
- For testing Stage, u must use sonarqube for testing your code quality

# Pipeline CI/CD

Disini saya melakukan installasi jenkins secara manual dengan mengikuti langkah-langkah yang ada pada dokumentasi ``` https://www.jenkins.io/doc/book/installing/linux/ ```

- Installasi Java
  ```
  sudo apt update
   sudo apt install fontconfig openjdk-17-jre
   java -version
   openjdk version "17.0.8" 2023-07-18

   **check sudah terinstal atau belum
   OpenJDK Runtime Environment (build 17.0.8+7-Debian-1deb12u1)
   OpenJDK 64-Bit Server VM (build 17.0.8+7-Debian-1deb12u1, mixed mode, sharing)
  ```

- Installasi Jenkins on Linux Ubuntu
  ```
   sudo wget -O /usr/share/keyrings/jenkins-keyring.asc \
      https://pkg.jenkins.io/debian-stable/jenkins.io-2023.key
   echo "deb [signed-by=/usr/share/keyrings/jenkins-keyring.asc]" \
      https://pkg.jenkins.io/debian-stable binary/ | sudo tee \
      /etc/apt/sources.list.d/jenkins.list > /dev/null
   sudo apt-get update
   sudo apt-get install jenkins
  ```

- Starting Jenkins
  ```
   sudo systemctl enable jenkins
   sudo systemctl start jenkins
   sudo systemctl status jenkins
  ```

# Staging CI/CD Pipeline

```
pipeline {
  agent any

  environment {
    SECRET = "${SECRET}"
    GATEWAY_URL = "${GATEWAY_URL}"
    DOCKER_SERVER = "${DOCKER_SERVER}"
    DOCKER_DIR_BACKEND = "${DOCKER_DIR_BACKEND}"
    GATEWAY_DIR = "${GATEWAY_DIR}"
    BRANCH = "${BRANCH}"
  }

  stages {
    stage (" Pull Repository Backend Dumbmerch ") {
      steps {
        sshagent(credentials: [SECRET]) {
          sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
          cd ${DOCKER_DIR_BACKEND}
          git pull origin ${BRANCH}
          echo "Git Pull Success"
          exit
          EOF"""
        }
      }
    }

    stage (" Image Build Backend ") {
      steps {
        sshagent(credentials: [SECRET]) {
          sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
          cd ${DOCKER_DIR_BACKEND}
          docker build --no-cache -t registry.olen.studentdumbways.my.id/be-dumbmerch:staging .
          echo "Docker Build Images Success"
          exit
          EOF"""
        }
      }
    }

    stage (" Code Analysis" ) {
      environment {
        scannerHome = tool name: 'sonarqube'
      }
      steps {
        script {
          withSonarQubeEnv('sonarqube') {
            sh """
              ${scannerHome}/bin/sonar-scanner \
              -Dsonar.projectKey=Backend_Dumbmerch \
              -Dsonar.sources=. \
              -Dsonar.host.url=${SONAR_URL} \
              -Dsonar.login=${SONAR_TOKEN}
            """
          }
        }
      }
    }

    stage (" Scan Images with Trivy ") {
      steps {
        sshagent(credentials: [SECRET]) {
          sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
          # Scan Docker image for vulnerabilities
          trivy image --severity CRITICAL,HIGH registry.olen.studentdumbways.my.id/be-dumbmerch:staging
          echo "Trivy Scan Success"
          exit
          EOF"""
        }
      }
    }

    stage (" Testing with wget spider ") {
      steps {
        sshagent(credentials: [SECRET]) {
          sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
          if wget --spider --server-response ${GATEWAY_URL} 2>&1 | grep -q "404 Not Found"; then
            echo "Backend berjalan"
          else 
              echo "Backend tidak berjalan"
              exit 1
          fi
          exit
          EOF"""
        }
      }
    }

    stage (" Push Images into Registry Private ") {
      steps {
        sshagent(credentials: [SECRET]) {
          sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
          cd ${DOCKER_DIR_BACKEND}
          docker push registry.olen.studentdumbways.my.id/be-dumbmerch:staging
          docker system prune -a -f
          echo "Docker Push Images Success"
          exit
          EOF"""
        }
      }
    }

    stage (" Pull and Re-deploy Backend using Docker Compose ") {
      steps {
        sshagent(credentials: [SECRET_GATEWAY]) {
          sh """ssh -o StrictHostKeyChecking=no ${GATEWAY_SERVER} << EOF
          cd ${GATEWAY_DIR}
          docker compose stop backend
          docker compose up -d backend
          echo "Pull images and Re-deploy Backend Success"
          exit
          EOF"""
        }
      }
    }
  }
}
```

# Production CI/CD Pipeline menggunakan Jenkins

- Membuat file Jenkinsfile pada repository Backend dan juga Frontend di branch production

```
pipeline {  
  agent any

  stages {
    stage (" Pull Repository Frontend Dumbmerch ") {
      steps {
        script {
          sshagent(credentials: [SECRET]) {
            sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
              cd ${DOCKER_DIR_FRONTEND}
              git checkout ${BRANCH_PROD}
              git pull origin ${BRANCH_PROD}
              exit
            EOF"""
          }
        }
      }
    }

    stage (" Image Build Frontend ") {
      steps {
        script {
          sshagent(credentials: [SECRET]) {
            sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
              cd ${DOCKER_DIR_FRONTEND}
              docker build --no-cache -t registry.olen.studentdumbways.my.id/fe-dumbmerch:production .
              echo "Docker Build Images Production Success"
            exit
            EOF"""
          }
        }
      }
    }

    stage (" Code Analysis ") {
      environment {
        scannerHome = tool name: 'sonarqube'
      }

      steps {
        script {
          withSonarQubeEnv('sonarqube') {
            sh """
              ${scannerHome}/bin/sonar-scanner \
              -Dsonar.projectKey=Frontend_Dumbmerch_Production \
              -Dsonar.sources=. \
              -Dsonar.host.url=${SONAR_URL} \
              -Dsonar.login=${SONAR_TOKEN}
            """
          }
        }
      }
    }

    stage (" Scan Images with Trivy ") {
      steps {
        script {
          sshagent(credentials: [SECRET]) {
            sh """
              # Scan Docker image for vulnerabilities
              trivy image --severity CRITICAL,HIGH registry.olen.studentdumbways.my.id/fe-dumbmerch:production
              echo "Trivy Scan Success"
            """
          }
        }
      }
    }

    stage (" Testing with wget spider ") {
      steps {
        script {
          sshagent(credentials: [SECRET]) {
            sh """
            if wget --spider --server-response ${APPS_URL_PROD} 2>&1 | grep -q "200 OK"; then
              echo "Frontend berjalan"
            else
                echo "Frontend tidak berjalan"
                exit 1
            fi
            """
          }
        }
      }
    }

    stage (" Push Images into Registry Private ") {
      steps {
        script {
          sshagent(credentials: [SECRET]) {
            sh """ssh -o StrictHostKeyChecking=no ${DOCKER_SERVER} << EOF
              cd ${DOCKER_DIR_FRONTEND}
              docker push registry.olen.studentdumbways.my.id/fe-dumbmerch:production
              docker system prune -a -f
              echo "Docker Push Images Success"
            exit
            EOF"""
          }
        }
      }
    }

    stage (" Re-deploy Frontend using Kubectl ") {
      steps {
        sshagent(credentials: [SECRET_KUBERNETES]) {
          sh """ssh -o StrictHostKeyChecking=no ${KUBERNETES_SERVER} << EOF
          cd ${KUBERNETES_DIR_FE}
          export KUBECONFIG=${KUBECONFIG}
          kubectl delete -f frontend.yaml
          kubectl apply -f frontend.yaml
          echo "Re-deploy Frontend Success"
          exit
          EOF"""
        }
      }
    }
  }
}
```

# Tahapan Pipeline CI/CD

- Mengambil repositori dari Github.
- Build docker image untuk backend/frontend.
- Memindai image menggunakan Trivy.
- Menguji aplikasi menggunakan wget spider.
- Push docker image ke registry private.
- Mendeploy ulang aplikasi backend/frontend.
